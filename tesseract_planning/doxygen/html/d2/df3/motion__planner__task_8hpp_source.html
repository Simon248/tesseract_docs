131 std::unique_ptr<TaskComposerNodeInfo> runImpl(TaskComposerContext& context,
132 OptionalTaskComposerExecutor /*executor*/ = std::nullopt) const override
133 {
134 auto info = std::make_unique<TaskComposerNodeInfo>(*this);
135 info->return_value = 0;
136 info->status_code = 0;
137
138 // --------------------
139 // Check that inputs are valid
140 // --------------------
141 auto env_poly = getData(*context.data_storage, INPUT_ENVIRONMENT_PORT);
142 if (env_poly.getType() != std::type_index(typeid(std::shared_ptr<const tesseract_environment::Environment>)))
143 {
144 info->status_code = 0;
145 info->status_message = "Input data '" + input_keys_.get(INPUT_ENVIRONMENT_PORT) + "' is not correct type";
146 CONSOLE_BRIDGE_logError("%s", info->status_message.c_str());
147 info->return_value = 0;
148 return info;
149 }
150
151 auto env = env_poly.template as<std::shared_ptr<const tesseract_environment::Environment>>();
152
153 auto input_data_poly = getData(*context.data_storage, INOUT_PROGRAM_PORT);
154 if (input_data_poly.getType() != std::type_index(typeid(CompositeInstruction)))
155 {
156 info->status_message = "Input instructions to MotionPlannerTask: " + name_ + " must be a composite instruction";
157 CONSOLE_BRIDGE_logError("%s", info->status_message.c_str());
158 return info;
159 }
160 tesseract_common::AnyPoly original_input_data_poly{ input_data_poly };
161
162 auto profiles =
163 getData(*context.data_storage, INPUT_PROFILES_PORT).template as<std::shared_ptr<ProfileDictionary>>();
164
165 // Make a non-const copy of the input instructions to update the start/end
166 auto& instructions = input_data_poly.template as<CompositeInstruction>();
167 if (instructions.getManipulatorInfo().empty())
168 throw std::runtime_error("Missing manipulator information");
169
170 // --------------------
171 // Fill out request
172 // --------------------
173 PlannerRequest request;
174 request.env = env;
175 request.instructions = instructions;
176 request.profiles = profiles;
177 request.format_result_as_input = format_result_as_input_;
178
179 // --------------------
180 // Fill out response
181 // --------------------
182 request.verbose = false;
183 if (console_bridge::getLogLevel() == console_bridge::LogLevel::CONSOLE_BRIDGE_LOG_DEBUG)
184 request.verbose = true;
185 PlannerResponse response = planner_->solve(request);
186
187 // --------------------
188 // Verify Success
189 // --------------------
190 if (response)
191 {
192 // Should only set on success to support error branching
193 setData(*context.data_storage, INOUT_PROGRAM_PORT, response.results);
194 info->return_value = 1;
195 info->color = "green";
196 info->status_code = 1;
197 info->status_message = response.message;
198 CONSOLE_BRIDGE_logDebug("Motion Planner process succeeded");
199 return info;
200 }
201
202 CONSOLE_BRIDGE_logInform("%s motion planning failed (%s) for process input: %s",
203 planner_->getName().c_str(),
204 response.message.c_str(),
205 instructions.getDescription().c_str());
206
207 // If the output key is not the same as the input key the output data should be assigned the input data for error
208 // branching
209 if (output_keys_.get(INOUT_PROGRAM_PORT) != input_keys_.get(INOUT_PROGRAM_PORT))
210 setData(*context.data_storage, INOUT_PROGRAM_PORT, original_input_data_poly);
211
212 info->status_message = response.message;
213 return info;
214 }
215};

60template <typename FloatType>
61PlannerResponse DescartesMotionPlanner<FloatType>::solve(const PlannerRequest& request) const
62{
63 PlannerResponse response;
64
65 // Get solver config
66 auto solver_profile =
67 getProfile<DescartesSolverProfile<FloatType>>(name_,
68 request.instructions.getProfile(name_),
69 *request.profiles,
70 std::make_shared<DescartesLadderGraphSolverProfile<FloatType>>());
71
72 auto solver = solver_profile->create();
73
74 std::vector<typename descartes_light::EdgeEvaluator<FloatType>::ConstPtr> edge_evaluators;
75 std::vector<typename descartes_light::WaypointSampler<FloatType>::ConstPtr> waypoint_samplers;
76 std::vector<typename descartes_light::StateEvaluator<FloatType>::ConstPtr> state_evaluators;
77
78 const tesseract_common::ManipulatorInfo& composite_mi = request.instructions.getManipulatorInfo();
79 if (composite_mi.empty())
80 throw std::runtime_error("Descartes, manipulator info is empty!");
81
82 // Flatten the input for planning
83 auto move_instructions = request.instructions.flatten(&moveFilter);
84
85 // Transform plan instructions into descartes samplers
86 int index = 0;
87 for (const auto& instruction : move_instructions)
88 {
89 assert(instruction.get().isMoveInstruction());
90 const auto& move_instruction = instruction.get().template as<MoveInstructionPoly>();
91
92 // Get Plan Profile
93 auto cur_plan_profile =
94 getProfile<DescartesPlanProfile<FloatType>>(name_,
95 move_instruction.getProfile(name_),
96 *request.profiles,
97 std::make_shared<DescartesDefaultPlanProfile<FloatType>>());
98
99 if (!cur_plan_profile)
100 throw std::runtime_error("DescartesMotionPlanner: Invalid profile");
101
102 if (move_instruction.getWaypoint().isJointWaypoint() &&
103 !move_instruction.getWaypoint().as<JointWaypointPoly>().isConstrained())
104 continue;
105
106 waypoint_samplers.push_back(cur_plan_profile->createWaypointSampler(move_instruction, composite_mi, request.env));
107 state_evaluators.push_back(cur_plan_profile->createStateEvaluator(move_instruction, composite_mi, request.env));
108 if (index != 0)
109 edge_evaluators.push_back(cur_plan_profile->createEdgeEvaluator(move_instruction, composite_mi, request.env));
110
111 ++index;
112 }
113
114 descartes_light::SearchResult<FloatType> descartes_result;
115 try
116 {
117 // Build Graph
118 if (!solver->build(waypoint_samplers, edge_evaluators, state_evaluators))
119 {
120 response.successful = false;
121 response.message = ERROR_FAILED_TO_BUILD_GRAPH;
122 return response;
123 }
124
125 // Search Graph
126 descartes_result = solver->search();
127 if (descartes_result.trajectory.empty())
128 {
129 CONSOLE_BRIDGE_logError("Search for graph completion failed");
130 response.successful = false;
131 response.message = ERROR_FAILED_TO_FIND_VALID_SOLUTION;
132 return response;
133 }
134 }
135 catch (...)
136 {
137 response.successful = false;
138 response.message = ERROR_FAILED_TO_BUILD_GRAPH;
139 return response;
140 }
141
142 response.data = std::move(solver);
143
144 // Get Manipulator Information
145 tesseract_kinematics::JointGroup::ConstPtr manip = request.env->getJointGroup(composite_mi.manipulator);
146 const std::vector<std::string> joint_names = manip->getJointNames();
147 const Eigen::MatrixX2d joint_limits = manip->getLimits().joint_limits;
148
149 // Enforce limits
150 std::vector<Eigen::VectorXd> solution{};
151 solution.reserve(descartes_result.trajectory.size());
152 for (const auto& js : descartes_result.trajectory)
153 {
154 solution.push_back(js->values.template cast<double>());
155 assert(tesseract_common::satisfiesLimits<double>(solution.back(), joint_limits));
156 tesseract_common::enforceLimits<double>(solution.back(), joint_limits);
157 }
158
159 // Flatten the results to make them easier to process
160 response.results = request.instructions;
161 auto results_instructions = response.results.flatten(&moveFilter);
162
163 // Loop over the flattened results and add them to response if the input was a plan instruction
164 std::size_t result_index{ 0 };
165 for (std::size_t idx = 0; idx < results_instructions.size(); idx++)
166 {
167 // If idx is zero then this should be the start instruction
168 assert((idx == 0) ? results_instructions.at(idx).get().isMoveInstruction() : true);
169 if (results_instructions.at(idx).get().isMoveInstruction())
170 {
171 auto& move_instruction = results_instructions.at(idx).get().as<MoveInstructionPoly>();
172 if (move_instruction.getWaypoint().isCartesianWaypoint())
173 {
174 assignSolution(move_instruction, joint_names, solution[result_index++], request.format_result_as_input);
175 }
176 else if (move_instruction.getWaypoint().isJointWaypoint())
177 {
178 auto& jwp = move_instruction.getWaypoint().as<JointWaypointPoly>();
179 if (jwp.isConstrained())
180 {
181 assignSolution(move_instruction, joint_names, solution[result_index++], request.format_result_as_input);
182 continue;
183 }
184
185 const Eigen::VectorXd& start_state = solution[result_index - 1];
186 Eigen::Index cnt = 1;
187 bool is_constrained = jwp.isConstrained();
188 while (!is_constrained)
189 {
190 auto temp = results_instructions.at(idx + static_cast<std::size_t>(cnt++)).get().as<MoveInstructionPoly>();
191 if (temp.getWaypoint().isCartesianWaypoint() || temp.getWaypoint().isStateWaypoint())
192 is_constrained = true;
193 else if (temp.getWaypoint().isJointWaypoint())
194 is_constrained = temp.getWaypoint().as<JointWaypointPoly>().isConstrained();
195 else
196 throw std::runtime_error("Unsupported Waypoint Type!");
197 }
198 const Eigen::VectorXd& end_state = solution[result_index];
199
200 Eigen::MatrixXd states = interpolate(start_state, end_state, cnt);
201 for (Eigen::Index i = 0; i < cnt - 1; ++i)
202 {
203 if (i != 0)
204 ++idx;
205 auto& interp_mi = results_instructions.at(idx).get().as<MoveInstructionPoly>();
206 assignSolution(interp_mi, joint_names, states.col(i + 1), request.format_result_as_input);
207 }
208 }
209 else if (move_instruction.getWaypoint().isStateWaypoint())
210 {
211 ++result_index;
212 }
213 else
214 {
215 throw std::runtime_error("Unsupported Waypoint Type!");
216 }
217 }
218 }
219
220 response.successful = true;
221 response.message = SOLUTION_FOUND;
222 return response;
223}
